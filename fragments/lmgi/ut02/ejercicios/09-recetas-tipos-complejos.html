2.9 Recetas: tipos complejos que usan otros tipos

Actividad para practicar un tipo complejo reutilizable que emplea tipos simples inventados previamente.

XSD de partida

<xs:schema xmlns:xs="http://www.w3.org/2001/XMLSchema">
  <xs:simpleType name="tipoCantidad">
    <xs:restriction base="xs:decimal">
      <xs:minExclusive value="0" />
    </xs:restriction>
  </xs:simpleType>

  <xs:simpleType name="tipoUnidad">
    <xs:restriction base="xs:string">
      <xs:enumeration value="g" />
      <xs:enumeration value="ml" />
      <xs:enumeration value="unidad" />
    </xs:restriction>
  </xs:simpleType>

  <xs:complexType name="tipoIngrediente">
    <xs:sequence>
      <xs:element name="nombre" type="xs:string" />
      <xs:element name="cantidad" type="tipoCantidad" />
      <xs:element name="unidad" type="tipoUnidad" />
    </xs:sequence>
    <xs:attribute name="opcional" type="xs:boolean" use="optional" />
  </xs:complexType>

  <xs:element name="receta">
    <xs:complexType>
      <xs:sequence>
        <xs:element name="nombre" type="xs:string" />
        <xs:element name="ingrediente" type="tipoIngrediente" minOccurs="1" maxOccurs="unbounded" />
      </xs:sequence>
    </xs:complexType>
  </xs:element>
</xs:schema>

Tarea

Genera un XML válido con una receta y al menos tres ingredientes. Usa el atributo opcional opcional en uno de ellos.
