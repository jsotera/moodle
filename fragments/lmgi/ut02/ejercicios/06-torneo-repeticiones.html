2.6 Torneo: repeticiones y estructuras anidadas

Actividad para practicar elementos repetibles, atributos y estructuras internas sin usar tipos reutilizables.

XSD de partida

<xs:schema xmlns:xs="http://www.w3.org/2001/XMLSchema">
  <xs:element name="torneo">
    <xs:complexType>
      <xs:sequence>
        <xs:element name="equipo" minOccurs="2" maxOccurs="8">
          <xs:complexType>
            <xs:sequence>
              <xs:element name="nombre" type="xs:string" />
              <xs:element name="jugador" minOccurs="2" maxOccurs="5">
                <xs:complexType>
                  <xs:sequence>
                    <xs:element name="alias">
                      <xs:simpleType>
                        <xs:restriction base="xs:string">
                          <xs:pattern value="[A-Za-z0-9_]+" />
                        </xs:restriction>
                      </xs:simpleType>
                    </xs:element>
                    <xs:element name="nivel">
                      <xs:simpleType>
                        <xs:restriction base="xs:integer">
                          <xs:minInclusive value="1" />
                          <xs:maxInclusive value="100" />
                        </xs:restriction>
                      </xs:simpleType>
                    </xs:element>
                  </xs:sequence>
                  <xs:attribute name="capitan" type="xs:boolean" use="optional" />
                </xs:complexType>
              </xs:element>
            </xs:sequence>
            <xs:attribute name="codigo" use="required">
              <xs:simpleType>
                <xs:restriction base="xs:string">
                  <xs:pattern value="EQ-[0-9][0-9]" />
                </xs:restriction>
              </xs:simpleType>
            </xs:attribute>
          </xs:complexType>
        </xs:element>
      </xs:sequence>
      <xs:attribute name="nombre" type="xs:string" use="required" />
    </xs:complexType>
  </xs:element>
</xs:schema>

Tarea

Genera un XML válido con al menos dos equipos. Cada equipo debe tener entre dos y cinco jugadores. Usa al menos un atributo opcional capitan.
