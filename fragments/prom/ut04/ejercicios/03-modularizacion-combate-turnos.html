4.3 Modularización: combate por turnos

Si, vamos a darle una vuelta a esto de modularizar código. Ahora tenéis una clase Codigo3. ¡Esta vez tiene más miga, leer los comentarios! Recordad, la tarea será reorganizarla en clases y métodos coherentes con nombres claros y responsables.

Código base a modularizar

Esta es la clase que vamos a usar como punto de partida:

public class Codigo3 {

    public int h1; // vida jugador1
    public int h2; // vida jugador2
    public int m1; // mana jugador1
    public int m2; // mana jugador2
    public int e1; // vida enemigo1
    public int e2; // vida enemigo2
    public int em1; // mana enemigo1
    public int em2; // mana enemigo2

    public void blablabla() {
        h1 = 100; m1 = 50;
        h2 = 100; m2 = 50;
        e1 = 80; em1 = 40;
        e2 = 80; em2 = 40;

        java.util.Random rnd = new java.util.Random();

        int turno = 1; // 1 = jugador1, 2 = jugador2

        for (int i = 0; i < 20; i++) {
            System.out.println("=== Nuevo turno ===");

            if (turno == 1 && h1 > 0) {
                System.out.println("Jugador 1: vida " + h1 + ", mana " + m1);
                int op = (int) (Math.random()*3)+1;
                int dmg = 5, cost = 0;
                if (op == 1) { dmg = 15; cost = 10; }
                if (op == 2) { dmg = 25; cost = 15; }
                if (op == 3) { dmg = 40; cost = 20; }

                if (m1 >= cost) {
                    m1 -= cost;
                    if (e1 > 0) e1 -= dmg;
                    else e2 -= dmg;
                    System.out.println("Jugador1 hace " + dmg + " daño");
                } else {
                    System.out.println("No tienes mana suficiente!");
                }
                turno = 2;
            } else if (turno == 2 && h2 > 0) {
                System.out.println("Jugador 2: vida " + h2 + ", mana " + m2);
                int op = (int) (Math.random()*3)+1;
                int dmg = 5, cost = 0;
                if (op == 1) { dmg = 15; cost = 10; }
                if (op == 2) { dmg = 25; cost = 15; }
                if (op == 3) { dmg = 40; cost = 20; }

                if (m2 >= cost) {
                    m2 -= cost;
                    if (e1 > 0) e1 -= dmg;
                    else e2 -= dmg;
                    System.out.println("Jugador2 hace " + dmg + " daño");
                } else {
                    System.out.println("No tienes mana suficiente!");
                }
                turno = 3;
            } else {
                if (e1 > 0) {
                    int op = (int) (Math.random()*3)+1;
                    int dmg = 5, cost = 0;
                    if (op == 1) { dmg = 10; cost = 5; }
                    if (op == 2) { dmg = 20; cost = 10; }
                    if (op == 3) { dmg = 30; cost = 15; }

                    if (em1 >= cost) {
                        em1 -= cost;
                        if (h1 > 0) h1 -= dmg;
                        else h2 -= dmg;
                        System.out.println("Enemigo1 hace " + dmg + " daño");
                    } else {
                        System.out.println("Enemigo1 no tiene mana suficiente");
                    }
                }

                if (e2 > 0) {
                    int op = (int) (Math.random()*3)+1;
                    int dmg = 5, cost = 0;
                    if (op == 1) { dmg = 10; cost = 5; }
                    if (op == 2) { dmg = 20; cost = 10; }
                    if (op == 3) { dmg = 30; cost = 15; }

                    if (em2 >= cost) {
                        em2 -= cost;
                        if (h1 > 0) h1 -= dmg;
                        else h2 -= dmg;
                        System.out.println("Enemigo2 hace " + dmg + " daño");
                    } else {
                        System.out.println("Enemigo2 no tiene mana suficiente");
                    }
                }
                turno = 1;
            }

            System.out.println("Estado -> J1: " + h1 + "/" + m1 + ", J2: " + h2 + "/" + m2 +
                    " | E1: " + e1 + "/" + em1 + ", E2: " + e2 + "/" + em2);
        }

        if (h1 > 0 || h2 > 0) System.out.println("¡Los jugadores han ganado!");
        else System.out.println("¡Los enemigos han ganado!");
    }
}
